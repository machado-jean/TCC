#            usando walk-forward com realimentação diária

# ==============================================================================
# ENTRADAS DO PIPELINE
# ==============================================================================
# df                    -> DataFrame com série temporal completa
# CONFIG                -> Dicionário de configurações globais
# FEATURES_EXOGENAS     -> Lista de variáveis exógenas do modelo
# MODELO                -> Classe do modelo (Baseline, Estatístico, ML ou DL)
# USA_OPTUNA            -> Bool indicando se usa Optuna
# RANDOM_STATE          -> Semente para reprodutibilidade

# ==============================================================================
# CONFIGURAÇÕES DERIVADAS
# ==============================================================================
# REGISTROS_POR_DIA     = 24 (para dados horários)
# REGISTROS_TESTE      = PERIODO_TESTE * REGISTROS_POR_DIA
# REGISTROS_LOOKAHEAD  = LOOKAHEAD * REGISTROS_POR_DIA
# HORIZONTE_REGISTROS  = HORIZONTE * REGISTROS_POR_DIA


# ==============================================================================
# PASSO 0 — PREPARAÇÃO INICIAL DOS DADOS
# ==============================================================================
# 1. Ler arquivo de dados
# 2. Converter coluna de datetime para datetime
# 3. Ordenar df por datetime crescente
# 4. Definir índice temporal
# 5. Logar início do pipeline

df = carregar_dados(CONFIG)
df = preparar_datetime(df, CONFIG)


# ==============================================================================
# PASSO 1 — PREPARAR TARGET E FEATURES EXÓGENAS
# ==============================================================================
# 1. y = df[COLUNA_TARGET]
# 2. X = df[FEATURES_EXOGENAS] (se existirem)
# 3. Se o modelo exigir normalização (LSTM, GRU, Deep Learning):
#    a. Normalizar y e X
#    b. Salvar scalers para inversão futura
# 4. Logar shapes e features utilizadas

y, X, scalers = preparar_target_e_exogenas(
    df,
    coluna_target=CONFIG["COLUNA_TARGET"],
    features_exogenas=FEATURES_EXOGENAS,
    normalizar=MODELO_REQUER_NORMALIZACAO
)


# ==============================================================================
# PASSO 2 — DIVIDIR TREINO E TESTE
# ==============================================================================
# 1. split_idx = len(y) - REGISTROS_TESTE
# 2. y_train = y[:split_idx]
# 3. y_test  = y[split_idx:]
# 4. X_train = X[:split_idx] (se existir)
# 5. X_test  = X[split_idx:] (se existir)
# 6. Logar shapes de treino e teste

y_train, y_test, X_train, X_test = split_treino_teste(
    y, X, CONFIG
)


# ==============================================================================
# PASSO 3 — OTIMIZAÇÃO DE HIPERPARÂMETROS (OPCIONAL)
# ==============================================================================
# Se USA_OPTUNA:
#   1. Definir espaço de busca
#   2. Para cada trial:
#       a. Instanciar modelo
#       b. Fit no treino
#       c. Prever janela de validação
#       d. Calcular métrica (RMSE)
#   3. Salvar study
#   4. Extrair melhores parâmetros
#
# Caso contrário:
#   PASSO 3 é ignorado

if USA_OPTUNA:
    study = carregar_ou_treinar_optuna(
        objective_func,
        y_train,
        X_train,
        CONFIG
    )
    best_params = study.best_params
else:
    best_params = PARAMETROS_FIXOS


# ==============================================================================
# PASSO 4 — TREINAR MODELO FINAL
# ==============================================================================
# 1. Instanciar modelo com best_params
# 2. Fit usando todo o conjunto de treino
# 3. Salvar modelo treinado (joblib / h5 / ckpt)

best_model = MODELO(**best_params)
best_model.fit(y_train, X_train)
salvar_modelo(best_model)


# ==============================================================================
# PASSO 5 — WALK-FORWARD COM REALIMENTAÇÃO
# ==============================================================================
# Inicializar:
#   y_preds = []
#   walk_forward_history = []
#
# Para cada dia i em [0, N_STEPS-1]:
#   1. start = i * REGISTROS_POR_DIA
#   2. end   = start + REGISTROS_POR_DIA
#   3. y_real_dia = y_test[start:end]
#   4. X_dia      = X_test[start:end] (se existir)
#   5. Prever:
#       - modelos univariados: predict(H)
#       - modelos multivariados: predict(H, X_dia)
#   6. Inverter escala (se aplicável)
#   7. Calcular métricas diárias
#   8. Armazenar histórico

y_pred_total, walk_forward_history = walk_forward(
    best_model,
    y_test,
    X_test,
    CONFIG,
    scalers
)


# ==============================================================================
# PASSO 6 — AVALIAÇÃO GLOBAL
# ==============================================================================
# 1. Concatenar previsões
# 2. Calcular métricas globais:
#    RMSE, MAE, MAPE, sMAPE, R²
# 3. Salvar métricas e histórico em JSON

metricas_globais = avaliar_global(
    walk_forward_history,
    CONFIG
)


# ==============================================================================
# PASSO 7 — VISUALIZAÇÃO DOS RESULTADOS
# ==============================================================================
# 1. (Opcional) Gráficos diários por modelo
# 2. Gráfico comparativo global entre modelos
# 3. Tabela consolidada de métricas

plotar_resultados(walk_forward_history)
plotar_comparativo_global()


# ==============================================================================
# PASSO 8 — ARMAZENAMENTO PARA COMPARAÇÃO ENTRE MODELOS
# ==============================================================================
# Salvar em lista global:
#   - nome do modelo
#   - métricas globais
#   - histórico de previsões

resultados_modelos.append({
    "modelo": NOME_MODELO,
    "metricas_globais": metricas_globais,
    "walk_forward_history": walk_forward_history
})
